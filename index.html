function doGet() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var rows = sheet.getDataRange().getValues(); var headers = rows[0]; var data = []; for (var i = 1; i < rows.length; i++) { var row = rows[i]; if (!row[1]) continue; // Bỏ qua dòng trống data.push({ id: row[0] || i, question: row[1], options: [row[2], row[3], row[4], row[5]], correct: (row[6] + '').trim().toUpperCase(), explanation: row[7] || '' }); } return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
